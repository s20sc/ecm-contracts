Here is a comparison of the code called in the image callback between ROS versions:
 * ROS2
```cpp
// Constructor
tf_buffer_ = std::make_shared<tf2_ros::Buffer>(this->get_clock());
tf_listener_ = std::make_shared<tf2_ros::TransformListener>(*tf_buffer_);

// Called from image callback:
try
{
  geometry_msgs::msg::TransformStamped tmp;
  tmp = tf_buffer_->lookupTransform("odom", "base_link", stamp, rclcpp::Duration::from_seconds(0.5));
}
catch(tf2::TransformException & ex)
{
  RCLCPP_ERROR(get_logger(), "%s (timeout=%.3fs)", ex.what(), 0.5);
}
```
 * ROS1
```cpp
// Constructor
tf::TransformListener tf_listener_;

// Called from image callback:
try
{
  std::string errorMsg;
  if(!tf_listener_.waitForTransform("odom", "base_link", stamp, ros::Duration(0.5), ros::Duration(0.01), &errorMsg))
  {
    ROS_ERROR("%s (timeout=%.3fs)", errorMsg.c_str(), 0.5);
    return;
  }

  tf::StampedTransform tmp;
  tf_listener_.lookupTransform("odom", "base_link", stamp, tmp);
}
catch(tf::TransformException & ex)
{
  ROS_ERROR("%s (timeout=%.3fs)", ex.what(), 0.5);
}
```

#### DDS Summary
~In conclusion, unless the way I am waiting for transform is wrong, I would recommend to use **rmw_cyclonedds_cpp** on ROS2 Humble.
