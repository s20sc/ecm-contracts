#include <cstdlib>
#include <iostream>

const char* str_or_null(const char* str)
{
  return str != nullptr ? str : "<nullptr>";
}

int main(int argc, char ** argv)
{
  (void) argc;
  (void) argv;

  auto* my_env_var = std::getenv("my_env_var");

  std::cerr << "my_env_var = '" << str_or_null(my_env_var) << "'" << std::endl;
  return 0;
}
